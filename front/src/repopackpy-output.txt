import React from 'react';
import { MapContainer, TileLayer } from 'react-leaflet';
import L from 'leaflet';
import RouteMenu from './RouteMenu';
import RadarOverlay from './Map/RadarOverlay';
import MapController from './Map/MapController';
import MapEventsHandler from './Map/MapEventsHandler';
import UserMarkers from './Map/UserMarkers';
import POIMarkers from './Map/POIMarkers';
import MapControls from './Map/MapControls';
import RoutingControl from './RoutingControl';
import { useMapRoutes } from '../hooks/useMapRoutes';

// Fix for default marker icons using CDN
const DefaultIcon = L.icon({
    iconUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon.png',
    shadowUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-shadow.png',
    iconSize: [25, 41],
    iconAnchor: [12, 41]
});

L.Marker.prototype.options.icon = DefaultIcon;

export default function MapCanvas() {
    const {
        markers,
        pois,
        isScanning,
        isLoading,
        radarCenter,
        route,
        lockedPoiIds,
        waypoints,
        setRoute,
        handleBuildRoute,
        handleAddMarker,
        handleMarkerMove,
        handleMarkerDelete,
        handleSelectSegment,
        handleToggleLock,
        handleClearRoute,
        handleRebuild,
        toggleRadar
    } = useMapRoutes();

    const handleDragStart = (e: React.DragEvent) => {
        e.dataTransfer.setData('text/plain', 'marker');
        e.dataTransfer.dropEffect = 'move';
    };

    return (
        <div id="map-root">
            <MapContainer
                center={[51.5335, 46.0342]}
                zoom={13}
                zoomControl={false}
                className="h-full w-full"
            >
                <TileLayer
                    attribution='&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors'
                    url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
                />

                <MapEventsHandler onMarkerAdd={handleAddMarker} />

                <MapController
                    active={isScanning}
                    center={radarCenter}
                    radius={500}
                />

                <UserMarkers
                    markers={route ? markers.filter(m => m.position[0] === route.points[0][0] && m.position[1] === route.points[0][1]) : markers}
                    onMarkerDelete={handleMarkerDelete}
                    onMarkerMove={handleMarkerMove}
                />

                <POIMarkers pois={route ? route.pois : pois} />

                {route && (
                    <RoutingControl
                        waypoints={waypoints}
                        onRouteFound={(distance) => {
                            setRoute(prev => {
                                if (!prev) return null;

                                // If we are looking at a specific segment, update its distance
                                if (prev.activeSegmentIndex != null && prev.activeSegmentIndex > 0) {
                                    const newSegmentDistances = [...(prev.segmentDistances || [])];
                                    newSegmentDistances[prev.activeSegmentIndex - 1] = distance;

                                    if (JSON.stringify(prev.segmentDistances) !== JSON.stringify(newSegmentDistances)) {
                                        return { ...prev, segmentDistances: newSegmentDistances };
                                    }
                                    return prev;
                                }

                                // If we are looking at the full route, update total distance
                                if (prev.totalDistance !== distance) {
                                    return { ...prev, totalDistance: distance };
                                }
                                return prev;
                            });
                        }}
                    />
                )}
            </MapContainer>

            <RouteMenu
                onBuildRoute={handleBuildRoute}
                activeRoute={route}
                onClearRoute={handleClearRoute}
                lockedPoiIds={lockedPoiIds}
                onToggleLock={handleToggleLock}
                onRebuild={handleRebuild}
                onSelectSegment={handleSelectSegment}
            />

            <RadarOverlay active={isScanning} radius={300} />

            <MapControls
                isLoading={isLoading}
                isScanning={isScanning}
                onToggleRadar={toggleRadar}
                onDragStart={handleDragStart}
            />
        </div>
    );
}

import { useEffect, useRef } from 'react';
import L from 'leaflet';
import 'leaflet-routing-machine';
import { useMap } from 'react-leaflet';

interface RoutingControlProps {
    waypoints: L.LatLng[];
    onRouteFound?: (distance: number, coordinates: [number, number][], legDistances?: number[]) => void;
}

const RoutingControl = ({ waypoints, onRouteFound }: RoutingControlProps) => {
    const map = useMap();
    const routingControlRef = useRef<L.Routing.Control | null>(null);

    useEffect(() => {
        if (!map || waypoints.length < 2) return;

        if (routingControlRef.current) {
            const currentWaypoints = routingControlRef.current.getWaypoints().map(w => w.latLng);
            const isSame = currentWaypoints.length === waypoints.length &&
                currentWaypoints.every((w, i) => w.lat === waypoints[i].lat && w.lng === waypoints[i].lng);

            if (!isSame) {
                routingControlRef.current.setWaypoints(waypoints);
            }
            return;
        }

        try {
            const routingControl = L.Routing.control({
                waypoints,
                routeWhileDragging: false,
                addWaypoints: false,
                fitSelectedRoutes: false, // Turn off auto-fit to avoid conflicts
                showAlternatives: false,
                lineOptions: {
                    styles: [
                        { color: '#ef4444', opacity: 0.6, weight: 6, dashArray: '10, 10' },
                        { color: '#ef4444', opacity: 1, weight: 2 }
                    ],
                    extendToWaypoints: true,
                    missingRouteTolerance: 10
                },
                router: L.Routing.osrmv1({
                    serviceUrl: 'https://router.project-osrm.org/route/v1',
                    profile: 'foot'
                }),
                // @ts-ignore
                createMarker: () => null,
            }).addTo(map);

            routingControlRef.current = routingControl;

            const container = routingControl.getContainer();
            if (container) {
                container.style.display = 'none';
            }

            routingControl.on('routesfound', (e) => {
                const routes = e.routes;
                if (routes && routes.length > 0) {
                    const mainRoute = routes[0];
                    const summary = mainRoute.summary;
                    const coordinates = mainRoute.coordinates.map((c: L.LatLng) => [c.lat, c.lng] as [number, number]);
                    const legDistances = mainRoute.instructions ?
                        [] : [];
                    if (waypoints.length > 0) {
                        const bounds = L.latLngBounds(coordinates);
                        map.fitBounds(bounds, { padding: [50, 50] });
                    }

                    if (onRouteFound) {
                        onRouteFound(summary.totalDistance, coordinates, legDistances);
                    }
                }
            });
        } catch (err) {
            console.error('Routing initialization error:', err);
        }
        return () => {
            if (routingControlRef.current) {
                const ctrl = routingControlRef.current;
                routingControlRef.current = null;
                setTimeout(() => {
                    try {
                        map.removeControl(ctrl);
                    } catch (err) {
                    }
                }, 0);
            }
        };
    }, [map, waypoints, onRouteFound]);

    return null;
};

export default RoutingControl;

import { useEffect, useRef } from 'react';
import { useMap } from 'react-leaflet';
import { RouteData } from '../../types';
import L from 'leaflet';

interface MapControllerProps {
  active: boolean;
  center: [number, number];
  radius: number;
  route: RouteData | null;
}

export default function MapController({ active, center, radius, route }: MapControllerProps) {
  const map = useMap();
  const lastRouteId = useRef<string | null>(null);

  useEffect(() => {
    if (active && center) {
      // Smoothly fly to normalized center
      map.flyTo(center, 15, { duration: 1.5 });
    }
  }, [active, center, map]);

  useEffect(() => {
    if (route && route.points.length > 0) {
      // Generate a simple ID to detect if it's a NEW route
      const routeId = JSON.stringify(route.pois.map(p => p.id));

      if (routeId !== lastRouteId.current) {
        lastRouteId.current = routeId;

        // Fit bounds smoothly
        const bounds = L.latLngBounds(route.points.map(p => [p[0], p[1]]));
        map.flyToBounds(bounds, {
          padding: [50, 50],
          duration: 1.5,
          easeLinearity: 0.1
        });
      }

      // If a segment is selected, zoom into it
      if (route.activeSegmentIndex != null) {
        const from = route.points[route.activeSegmentIndex - 1] || route.points[0];
        const to = route.points[route.activeSegmentIndex];
        const segBounds = L.latLngBounds([[from[0], from[1]], [to[0], to[1]]]);
        map.flyToBounds(segBounds, {
          padding: [80, 80],
          duration: 1.2
        });
      }
    }
  }, [route, map]);

  return null;
}

import { useEffect } from 'react';
import { useMap } from 'react-leaflet';

interface MapEventsHandlerProps {
  onMarkerAdd: (pos: [number, number]) => void;
}

export default function MapEventsHandler({ onMarkerAdd }: MapEventsHandlerProps) {
  const map = useMap();
  
  useEffect(() => {
    const container = map.getContainer();
    
    const handleDragOver = (e: DragEvent) => {
      e.preventDefault();
      if (e.dataTransfer) {
        e.dataTransfer.dropEffect = 'move';
      }
    };
    
    const handleDrop = (e: DragEvent) => {
      e.preventDefault();
      const latlng = map.mouseEventToLatLng(e);
      onMarkerAdd([latlng.lat, latlng.lng]);
    };
    
    container.addEventListener('dragover', handleDragOver);
    container.addEventListener('drop', handleDrop);
    
    return () => {
      container.removeEventListener('dragover', handleDragOver);
      container.removeEventListener('drop', handleDrop);
    };
  }, [map, onMarkerAdd]);

  return null;
}

import { useState, useCallback, useEffect, useMemo } from 'react';

import L from 'leaflet';

import { UserMarker, POI, RouteData } from '../types';



// Haversine distance in meters

function getDistance(pos1: [number, number], pos2: [number, number]) {

    const R = 6371e3; // meters

    const φ1 = pos1[0] * Math.PI / 180;

    const φ2 = pos2[0] * Math.PI / 180;

    const Δφ = (pos2[0] - pos1[0]) * Math.PI / 180;

    const Δλ = (pos2[1] - pos1[1]) * Math.PI / 180;



    const a = Math.sin(Δφ / 2) * Math.sin(Δφ / 2) +

        Math.cos(φ1) * Math.cos(φ2) *

        Math.sin(Δλ / 2) * Math.sin(Δλ / 2);

    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));



    return R * c;

}



export function useMapRoutes() {

    const [markers, setMarkers] = useState<UserMarker[]>([]);

    const [pois, setPois] = useState<POI[]>([]);

    const [isScanning, setIsScanning] = useState(false);

    const [isLoading, setIsLoading] = useState(false);

    const [radarCenter, setRadarCenter] = useState<[number, number]>([51.5335, 46.0342]);

    const [route, setRoute] = useState<RouteData | null>(null);

    const [lockedPoiIds, setLockedPoiIds] = useState<Set<number>>(new Set());

    const [lastRouteSettings, setLastRouteSettings] = useState<{ categories: string[], minSteps: number, maxSteps: number, isReturnBack: boolean } | null>(null);



    const fetchPOIs = useCallback(async (center: [number, number]) => {

        setIsLoading(true);

        try {

            const response = await fetch(`/poi/nearby/recommended?lat=${center[0]}&lng=${center[1]}&limit=50`);

            if (response.ok) {

                const data = await response.json();

                setPois(data);

            } else {

                throw new Error('Response not ok');

            }

        } catch (error) {

            console.error('Failed to fetch POIs:', error);

            // Fallback mocked data

            setPois([

                {

                    id: 999,

                    name: "Тестовая точка",

                    address: "Улица Тестовая, 1",

                    coordinates: [center[1] + 0.005, center[0] + 0.005],

                    category: "culture",

                    score: 0.95,

                    predicted_rating: 5,

                    model_confidence: 0.8,

                    review_count: 1

                },

                {

                    id: 998,

                    name: "Парк Отдыха",

                    address: "Улица Зеленая, 10",

                    coordinates: [center[1] - 0.005, center[0] + 0.005],

                    category: "natural_outdoor",

                    score: 0.92,

                    predicted_rating: 4,

                    model_confidence: 0.85,

                    review_count: 20

                },

                {

                    id: 997,

                    name: "Кафе Уют",

                    address: "Проспект Мира, 5",

                    coordinates: [center[1] + 0.005, center[0] - 0.005],

                    category: "food_drink",

                    score: 0.88,

                    predicted_rating: 4.5,

                    model_confidence: 0.9,

                    review_count: 50

                }

            ]);

        } finally {

            setIsLoading(false);

            setTimeout(() => setIsScanning(false), 3000);

        }

    }, []);



    const handleBuildRoute = useCallback(async (settings: { categories: string[], minSteps: number, maxSteps: number, isReturnBack: boolean }) => {

        if (pois.length === 0) return;

        setLastRouteSettings(settings);



        // Filter POIs by categories

        const filteredPois = pois.filter(p => {

            return settings.categories.some(cat => p.category.toLowerCase().includes(cat.toLowerCase()));

        });



        if (filteredPois.length < 2 && lockedPoiIds.size === 0) {

            alert("Недостаточно точек для построения маршрута. Попробуйте выбрать больше категорий.");

            return;

        }



        setIsLoading(true);



        try {

            const lockedPois = route ? route.pois.filter(p => lockedPoiIds.has(p.id)) : [];

            const selectedPois: POI[] = [...lockedPois];



            const currentUnlockedIds = route ? route.pois.filter(p => !lockedPoiIds.has(p.id)).map(p => p.id) : [];



            const categoryCounts = new Map<string, number>();

            selectedPois.forEach(p => {

                const cat = p.category.toLowerCase();

                categoryCounts.set(cat, (categoryCounts.get(cat) || 0) + 1);

            });



            const sorted = [...filteredPois]

                .filter(p => !selectedPois.find(sp => sp.id === p.id) && !currentUnlockedIds.includes(p.id))

                .sort((a, b) => b.score - a.score);



            const avgSteps = (settings.minSteps + settings.maxSteps) / 2;

            const targetMeters = avgSteps * 0.7;

            const targetPoiCount = Math.max(2, Math.min(8, Math.round(targetMeters / 600)));



            for (const poi of sorted) {

                if (selectedPois.length >= targetPoiCount) break;



                const cat = poi.category.toLowerCase();

                const currentCount = categoryCounts.get(cat) || 0;



                if (cat.includes('food_drink') && currentCount >= 2) continue;



                if (currentCount === 0) {

                    selectedPois.push(poi);

                    categoryCounts.set(cat, 1);

                }

            }



            if (selectedPois.length < targetPoiCount) {

                for (const poi of sorted) {

                    if (selectedPois.length >= targetPoiCount) break;

                    if (selectedPois.find(p => p.id === poi.id)) continue;



                    const cat = poi.category.toLowerCase();

                    const currentCount = categoryCounts.get(cat) || 0;



                    if (cat.includes('food_drink') && currentCount >= 2) continue;



                    selectedPois.push(poi);

                    categoryCounts.set(cat, currentCount + 1);

                }

            }



            if (selectedPois.length < 2) {

                const fallbackPool = filteredPois.filter(p => !selectedPois.find(sp => sp.id === p.id));

                for (const poi of fallbackPool) {

                    if (selectedPois.length >= targetPoiCount) break;

                    selectedPois.push(poi);

                }

            }



            const startPoint: [number, number] = markers.length > 0 ? markers[markers.length - 1].position : radarCenter;



            const remaining = [...selectedPois];

            const orderedPois: POI[] = [];

            const waypointChain: [number, number][] = [startPoint];

            let currentPos = startPoint;



            while (remaining.length > 0) {

                let closestIdx = 0;

                let minD = Infinity;

                remaining.forEach((p, idx) => {

                    const d = getDistance(currentPos, [p.coordinates[1], p.coordinates[0]]);

                    if (d < minD) {

                        minD = d;

                        closestIdx = idx;

                    }

                });

                const nextPoi = remaining.splice(closestIdx, 1)[0];

                orderedPois.push(nextPoi);

                const nextCoords: [number, number] = [nextPoi.coordinates[1], nextPoi.coordinates[0]];

                waypointChain.push(nextCoords);

                currentPos = nextCoords;

            }



            if (settings.isReturnBack) {

                waypointChain.push(startPoint);

            }



            setRoute({

                points: waypointChain,

                segments: [],

                pois: orderedPois,

                totalDistance: 0,

                isReturnBack: settings.isReturnBack,

                activeSegmentIndex: null,

                segmentDistances: new Array(orderedPois.length + (settings.isReturnBack ? 1 : 0)).fill(0)

            });



        } catch (error) {

            console.error('Routing error:', error);

            alert("Не удалось построить маршрут по дорогам.");

        } finally {

            setIsLoading(false);

        }

    }, [pois, lockedPoiIds, route, markers, radarCenter]);



    useEffect(() => {

        if (isScanning) {

            fetchPOIs(radarCenter);

        }

    }, [isScanning, radarCenter, fetchPOIs]);



    const handleAddMarker = useCallback((position: [number, number]) => {

        const newMarker: UserMarker = {

            id: Math.random().toString(36).substring(7),

            position,

            title: 'Стартовая точка'

        };

        setMarkers([newMarker]);

        setRadarCenter(position);

    }, []);



    const handleMarkerMove = useCallback((id: string, newPos: [number, number]) => {

        setMarkers(prev => prev.map(m =>

            m.id === id ? { ...m, position: newPos } : m

        ));

        setRadarCenter(newPos);

    }, []);



    const handleMarkerDelete = useCallback((id: string) => {

        setMarkers(prev => prev.filter(m => m.id !== id));

    }, []);



    const handleSelectSegment = useCallback((index: number | null) => {

        setRoute(prev => prev ? { ...prev, activeSegmentIndex: index } : null);

    }, []);



    const handleToggleLock = useCallback((id: number) => {

        setLockedPoiIds(prev => {

            const next = new Set(prev);

            if (next.has(id)) next.delete(id);

            else next.add(id);

            return next;

        });

    }, []);



    const handleClearRoute = useCallback(() => {

        setRoute(null);

        setLockedPoiIds(new Set());

    }, []);



    const handleRebuild = useCallback(() => {

        if (lastRouteSettings) {

            handleBuildRoute(lastRouteSettings);

        }

    }, [lastRouteSettings, handleBuildRoute]);



    const toggleRadar = useCallback(() => {

        if (isScanning) return;

        setIsScanning(true);

        if (markers.length > 0) {

            setRadarCenter(markers[markers.length - 1].position);

        }

    }, [isScanning, markers]);



    const waypoints = useMemo(() => {

        if (!route) return [];

        if (route.activeSegmentIndex != null && route.activeSegmentIndex > 0) {

            const from = route.points[route.activeSegmentIndex - 1];

            const to = route.points[route.activeSegmentIndex];

            return [L.latLng(from[0], from[1]), L.latLng(to[0], to[1])];

        }

        return route.points.map(p => L.latLng(p[0], p[1]));

    }, [route]);



    return {

        markers,

        pois,

        isScanning,

        isLoading,

        radarCenter,

        route,

        lockedPoiIds,

        waypoints,

        setRoute,

        handleBuildRoute,

        handleAddMarker,

        handleMarkerMove,

        handleMarkerDelete,

        handleSelectSegment,

        handleToggleLock,

        handleClearRoute,

        handleRebuild,

        toggleRadar

    };

}
