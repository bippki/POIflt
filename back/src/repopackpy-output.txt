import { Controller, Get, Query } from '@nestjs/common';
import { PoiService } from '../service/poi.service';

@Controller('poi')
export class PoiController {
    constructor(private readonly poiService: PoiService) { }

    @Get('nearby')
    async findNearby(
        @Query('lat') lat: number,
        @Query('lng') lng: number,
    ) {
        return await this.poiService.findNearbyRaw(lat, lng);
    }

    @Get('nearby/recommended')
    async findRecommended(
        @Query('lat') lat: number,
        @Query('lng') lng: number,
        @Query('limit') limit: number = 5,
    ) {
        return await this.poiService.getRecommendedNearby(lat, lng, limit);
    }
}

import { Entity, Column, PrimaryGeneratedColumn, OneToMany } from 'typeorm';
import { Review } from './review.entity';

@Entity()
export class Poi {
    @PrimaryGeneratedColumn()
    id!: number;

    @Column()
    name!: string;

    @Column()
    address!: string;

    @Column({ type: 'geography', spatialFeatureType: 'Point', srid: 4326 })
    geom!: string;

    @Column()
    category!: string;

    @OneToMany(() => Review, (review) => review.poi)
    reviews!: Review[];
}

import { Entity, Column, PrimaryGeneratedColumn, ManyToOne, JoinColumn } from 'typeorm';
import { Poi } from './poi.entity';

@Entity('reviews')
export class Review {
    @PrimaryGeneratedColumn()
    id: number;

    @Column()
    poi_id: number;

    @Column({ type: 'numeric', nullable: true })
    rating?: number; // Может быть null

    @Column({ nullable: true })
    text?: string; // Может быть null

    @ManyToOne(() => Poi, (poi) => poi.reviews)
    @JoinColumn({ name: 'poi_id' })
    poi: Poi;
}

import { HttpException, HttpStatus, Injectable, Logger } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { HttpService } from '@nestjs/axios';
import { firstValueFrom } from 'rxjs';
import { Poi } from '../entity/poi.entity';
import { AxiosError } from 'axios';

@Injectable()
export class PoiService {
    private readonly logger = new Logger(PoiService.name);

    constructor(
        @InjectRepository(Poi)
        private readonly poiRepo: Repository<Poi>,
        private readonly httpService: HttpService,
    ) { }

    async findNearbyRaw(lat: number, lng: number) {
        return await this.poiRepo
            .createQueryBuilder('poi')
            .leftJoinAndSelect('poi.reviews', 'reviews')
            .where(
                `ST_DWithin(poi.geom, ST_SetSRID(ST_MakePoint(:lng, :lat), 4326), 1000)`,
                { lat, lng },
            )
            .getMany();
    }

    async getRecommendedNearby(lat: number, lng: number, limit: number) {
        const pois = await this.findNearbyRaw(lat, lng);
        if (pois.length === 0) return [];

        try {
            const { data } = await firstValueFrom(
                this.httpService.post('http://localhost:8000/recommend', {
                    pois: pois,
                    top_k: limit,
                    preferred_rubrics: [],
                }),
            );
            return data;
        } catch (error: unknown) {
            const errorMessage = error instanceof AxiosError
                ? error.response?.data?.message || error.message
                : 'Unknown error';

            this.logger.error(`AI Service failure: ${errorMessage}`);

            throw new HttpException(
                {
                    status: HttpStatus.SERVICE_UNAVAILABLE,
                    error: 'AI Recommendation Service is currently unavailable',
                    details: errorMessage,
                },
                HttpStatus.SERVICE_UNAVAILABLE,
            );
        }
    }
}
